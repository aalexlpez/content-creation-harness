<script src="./lib/gsap.min.js"></script>
<script>
  // ── Anclas (s desde el inicio de la pieza). Estimadas con el SRT de YouTube, toma de
  //    6:43–7:00; se recalculan con la toma elegida. La pieza entra 0,3 s antes de "Aquí".
  const A = {
    IN: 0.0,        // entrada deslizante de 0,3 s (whoosh)
    TYPE: 0.9,      // "eliges un propósito"
    NEXT1: 2.3,     // toque en Siguiente
    S2: 2.7,        // entra el paso 2 con el mes en "enero"
    FEB: 3.4,       // "febrero": cambia el mes y la pantalla pasa a gris (600 ms)
    WHY: 4.0,       // "eliges por qué": entran los seis obstáculos
    TAP: 6.5,       // "no tuve tiempo": toque y avance automático
    S3: 7.1,        // entra el paso 3
    SI: 10.3,       // "si llego a casa…": se escribe la situación (gris → 50 %)
    ACC: 12.5,      // "hago 5 minutos…": se escribe la acción (gris → 0 %)
    VER: 14.8,      // toque en "Ver mi plan" (continúa MG-07)
  };
  const SCALE = 1.3385;

  // Divide un texto en letras para "escribirlo" con opacidad escalonada
  function split(id) {
    const el = document.getElementById(id), t = el.textContent; el.textContent = "";
    for (const ch of t) { const s = document.createElement("span"); s.className = "c"; s.textContent = ch; el.appendChild(s); }
    return el.querySelectorAll(".c");
  }
  const c1 = split("t1"), c2 = split("t2"), c3 = split("t3");
  const type = (chars, t0, dur) => tl.to(chars, { opacity: 1, duration: 0.001, stagger: dur / chars.length, ease: "none" }, t0);

  // Centro de un elemento de la app, en coordenadas de la app (390×689)
  function center(el) {
    const a = document.getElementById("app").getBoundingClientRect(), r = el.getBoundingClientRect();
    return { x: (r.left - a.left + r.width / 2) / SCALE, y: (r.top - a.top + r.height / 2) / SCALE };
  }
  function tapAt(el, t) {
    const p = center(el);
    tl.set(["#tap", "#ring"], { left: p.x, top: p.y, scale: 0.6, opacity: 0 }, t - 0.12);
    tl.to("#tap", { opacity: 1, scale: 1, duration: 0.12, ease: "power2.out" }, t - 0.12);
    tl.to("#ring", { opacity: 1, duration: 0.05 }, t);
    tl.to("#ring", { scale: 1.8, opacity: 0, duration: 0.35, ease: "power2.out" }, t + 0.05);
    tl.to("#tap", { opacity: 0, duration: 0.2, ease: "power1.in" }, t + 0.12);
    tl.to(el, { scale: 0.97, duration: 0.08, ease: "power2.out", yoyo: true, repeat: 1 }, t - 0.02);
  }
  const stepIn = (id, t) => { tl.fromTo(id, { x: 28, opacity: 0 }, { x: 0, opacity: 1, duration: 0.32, ease: "expo.out" }, t); };
  const stepOut = (id, t) => { tl.to(id, { x: -28, opacity: 0, duration: 0.22, ease: "power2.in" }, t); };

  const tl = gsap.timeline({ paused: true });
  gsap.set("#mFeb", { yPercent: 100, opacity: 0 });
  gsap.set(".ob", { scale: 0.86, opacity: 0 });
  gsap.set("#phone", { transformOrigin: "50% 50%" });

  // Entrada deslizante del celular (0,3 s) y paso 1 visible
  tl.fromTo("#phone", { y: 120, opacity: 0 }, { y: 0, opacity: 1, duration: 0.3, ease: "expo.out" }, A.IN);
  tl.fromTo("#mockLabel", { opacity: 0 }, { opacity: 1, duration: 0.3 }, A.IN + 0.1);
  tl.set("#s1", { opacity: 1 }, A.IN);

  // Paso 1: se escribe el propósito
  type(c1, A.TYPE, 1.1);
  tapAt(document.getElementById("b1"), A.NEXT1);
  stepOut("#s1", A.NEXT1 + 0.15);

  // Paso 2: entra con "enero"; en "febrero" cambia el mes, zoom al mes y gris en 600 ms
  stepIn("#s2", A.S2);
  tl.to("#mEne", { yPercent: -100, opacity: 0, duration: 0.3, ease: "power2.inOut" }, A.FEB);
  tl.to("#mFeb", { yPercent: 0, opacity: 1, duration: 0.3, ease: "power2.inOut" }, A.FEB);
  tl.to("#app", { "--g": 1, duration: 0.6, ease: "power1.inOut" }, A.FEB);
  // cámara virtual: 1,10× hacia la etiqueta de mes y regreso (0,6 s)
  tl.set("#phone", { transformOrigin: "30% 18%" }, A.FEB - 0.01);
  tl.to("#phone", { scale: 1.10, duration: 0.3, ease: "power2.out" }, A.FEB - 0.05)
    .to("#phone", { scale: 1, duration: 0.45, ease: "power2.inOut" }, A.FEB + 0.55);
  // seis obstáculos escalonados (120 ms), con resorte
  tl.to(".ob", { scale: 1, opacity: 1, duration: 0.42, ease: "back.out(1.7)", stagger: 0.12 }, A.WHY);
  // toque en "No tuve tiempo": se marca y avanza solo
  tapAt(document.getElementById("ob0"), A.TAP);
  tl.set("#ob0", { className: "ob sel" }, A.TAP + 0.05);
  stepOut("#s2", A.S3 - 0.2);

  // Paso 3: gris al 100 %; la situación lo baja al 50 % y la acción al 0 %
  stepIn("#s3", A.S3);
  tl.set("#f2", { className: "field active" }, A.SI - 0.1);
  type(c2, A.SI, 1.4);
  tl.to("#app", { "--g": 0.5, duration: 0.4, ease: "power1.inOut" }, A.SI + 1.4);
  tl.set("#f2", { className: "field" }, A.ACC - 0.1);
  tl.set("#f3", { className: "field active" }, A.ACC - 0.1);
  type(c3, A.ACC, 2.0);
  tl.to("#app", { "--g": 0, duration: 0.4, ease: "power1.inOut" }, A.ACC + 2.0);
  tl.set("#f3", { className: "field" }, A.VER - 0.2);
  // zoom suave 1,06× sobre el campo activo
  tl.set("#phone", { transformOrigin: "50% 40%" }, A.SI - 0.3);
  tl.to("#phone", { scale: 1.06, duration: 0.6, ease: "power1.inOut" }, A.SI - 0.3)
    .to("#phone", { scale: 1, duration: 0.6, ease: "power1.inOut" }, A.VER - 0.7);
  tapAt(document.getElementById("b3"), A.VER);

  window.__timelines = window.__timelines || {};
  window.__timelines["root"] = tl;
</script>
