<script>
  // ── Anclas (s). Estimadas con el SRT de YouTube, toma de 7:05–7:17 ("Con esto te
  //    llevas…" empieza ≈0,6 s después del inicio de la pieza). Se recalculan con la toma.
  const A = {
    IN: 0.0,      // el resultado sustituye al paso 3 (mismo celular)
    SEAL: 0.4,    // el sello cae justo antes de "Con esto te llevas"
    CAL: 2.5,     // "calendario": sube el bloque Recuérdamelo y se marcan las 20 citas
    FONDO: 3.9,   // "fondo de pantalla": el celular pasa a la pantalla de bloqueo
    OUT: 5.4,     // sale deslizándose hacia abajo; vuelve la cámara
  };

  // 4 semanas × 7 días; de lunes a viernes = 20 apariciones
  const cells = document.getElementById("cells"), marks = [];
  for (let w = 0; w < 4; w++) for (let d = 0; d < 7; d++) {
    const c = document.createElement("div"); c.className = "cell"; const i = document.createElement("i"); c.appendChild(i); cells.appendChild(c);
    if (d < 5) marks.push(i);
  }

  const tl = gsap.timeline({ paused: true });
  gsap.set("#seal", { rotation: -6, scale: 1.4, opacity: 0, transformOrigin: "50% 50%" });
  gsap.set("#cal", { y: 40, opacity: 0 });

  // 1 · El resultado entra (continúa a MG-06)
  tl.fromTo("#result", { x: 28, opacity: 0 }, { x: 0, opacity: 1, duration: 0.32, ease: "expo.out" }, A.IN);

  // 2 · Sello: escala 1,4 → 1 en 300 ms a −6° y un pequeño golpe del celular
  tl.to("#seal", { opacity: 1, duration: 0.06, ease: "none" }, A.SEAL);
  tl.to("#seal", { scale: 1, duration: 0.3, ease: "power3.in" }, A.SEAL);
  tl.to("#phone", { y: 6, duration: 0.06, ease: "power2.out" }, A.SEAL + 0.3)
    .to("#phone", { y: -2, duration: 0.1, ease: "power2.inOut" })
    .to("#phone", { y: 0, duration: 0.14, ease: "power2.out" });

  // 3 · Calendario: sube el bloque y se marcan las 20 apariciones (40 ms entre celdas)
  tl.to("#cal", { y: 0, opacity: 1, duration: 0.4, ease: "expo.out" }, A.CAL);
  tl.to(marks, { opacity: 1, duration: 0.08, ease: "none", stagger: 0.04 }, A.CAL + 0.25);

  // 4 · Fondo de pantalla: el mismo celular pasa a la pantalla de bloqueo (escala y fundido)
  tl.set("#screen", { backgroundColor: "#171421" }, A.FONDO);
  tl.to("#result", { scale: 1.3385 * 0.94, opacity: 0, duration: 0.25, ease: "power2.in" }, A.FONDO);
  tl.fromTo("#lock", { opacity: 0, scale: 1.3385 * 1.04 }, { opacity: 1, scale: 1.3385, duration: 0.38, ease: "power2.out" }, A.FONDO + 0.2);

  // 5 · Salida hacia abajo
  tl.to("#phone", { y: 260, opacity: 0, duration: 0.3, ease: "power2.in" }, A.OUT);
  tl.to("#mockLabel", { opacity: 0, duration: 0.25 }, A.OUT);

  window.__timelines = window.__timelines || {};
  window.__timelines["root"] = tl;
</script>
