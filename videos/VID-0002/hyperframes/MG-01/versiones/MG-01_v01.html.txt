<script>
  // Marcas de tiempo (s). Cambian al ajustar a la voz real:
  //   T_FEB  = instante en que la hoja pasa a FEBRERO (la palabra "febrero")
  //   T_ENE  = instante en que vuelve a ENERO ("todavía no")
  const T_IN = 0.0, T_FEB = 0.9, T_ENE = 4.4, T_OUT = 5.2, FLIP = 0.17;

  const tl = gsap.timeline({ paused: true });
  gsap.set("#pFebrero", { scaleY: 0 });
  gsap.set("#sheet", { transformOrigin: "50% 0%", rotation: -20, y: -230, opacity: 0 });

  // Entrada con resorte desde arriba: -20° -> -6°
  tl.to("#sheet", { y: 0, rotation: -6, opacity: 1, duration: 0.5, ease: "back.out(1.5)" }, T_IN);

  // ENERO -> FEBRERO: la hoja gira sobre su borde superior
  tl.to("#pEnero",   { scaleY: 0, duration: FLIP, ease: "power2.in"  }, T_FEB);
  tl.to("#pFebrero", { scaleY: 1, duration: FLIP, ease: "power2.out" }, T_FEB + FLIP);

  // FEBRERO -> ENERO
  tl.to("#pFebrero", { scaleY: 0, duration: FLIP, ease: "power2.in"  }, T_ENE);
  tl.to("#pEnero",   { scaleY: 1, duration: FLIP, ease: "power2.out" }, T_ENE + FLIP);

  // Salida: fundido de 0,3 s
  tl.to("#sheet", { opacity: 0, y: -30, duration: 0.3, ease: "power2.in" }, T_OUT);

  window.__timelines = window.__timelines || {};
  window.__timelines["root"] = tl;
</script>
